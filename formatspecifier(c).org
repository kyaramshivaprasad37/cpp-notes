#+title: Formatspecifier

Special tokens that begin with a % sysbol,
followed by a character that specifies the data type
and optional modifiers(width,precision,flags).
They control how data is displayed or interpreted.

#+begin_src C :results output
#include <stdio.h>

int main() {

    int age = 25;
    float price = 19.99;
    double pi = 3.141592;
    char currency = '$';
    char name[] = "Shiva";

    printf("%d\n",age);
    printf("%f\n",price);
    printf("%lf\n",pi);
    printf("%c\n",currency);
    printf("%s\n",name);

    return 0;
}
#+end_src

#+RESULTS:
: 25
: 19.990000
: 3.141592
: $
: Shiva

#+begin_src C :results output
#include <stdio.h>

int main() {

    float price1 = 19.99;
    float price2 = 1.50;
    float price3 = -100.00;

    printf("%.3f\n",price1);
    printf("%.1f\n",price2);
    printf("%.1f\n",price3);

    printf("%+7.2f\n",price1);
    printf("%+7.2f\n",price2);
    printf("%+7.2f\n",price3);
    return 0 ;
}
