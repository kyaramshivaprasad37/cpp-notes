        a[i] = 2;
    }

        cout << a[i] << ' ';
    }
    return 0;
}

#include <iostream>
using namespace std;

int main() {
    int a[] = {0,1,2,0,1,2,1,2,0,0,1};
    int n = sizeof(a)/sizeof(a[0]);
    int mid = 0;
    int low = 0;
    int high = n-1;

    while(mid <= high) {
        if(a[mid] == 0) {
            swap(a[mid], a[low]);
            mid++;
            low++;
        }else if(a[mid] == 1) {
            mid++;
        }else{
            swap(a[mid], a[high]);
            high--;
        }
    }
    for(int i =0;i < n; i++) {
        cout << a[i] << ' ';
    }
    return 0;
}

#+begin_src cpp :results output
#include <iostream>

// 1. The Blueprints (Structures)
struct DataNode {
    int data;
    DataNode* next;
};

struct HeaderNode {
    int size;
    DataNode* next;
};

// 2. Initialize the Empty List
HeaderNode* createCircularHeaderList() {
    HeaderNode* h = new HeaderNode();
    h->size = 0;
    h->next = (DataNode*)h; // Self-loop using the cast
    return h;
}

// 3. Check if Empty
bool isEmpty(HeaderNode* h) {
    return h->next == (DataNode*)h;
}

// 4. Insert at Beginning (Topic 1)
void insertAtBeginning(HeaderNode* h, int value) {
    DataNode* newNode = new DataNode();
    newNode->data = value;

    // Point new node to the current first data node (or back to header if empty)
    newNode->next = h->next;

    // Header points to the new node
    h->next = newNode;

    // Increment the metadata size counter
    h->size++;
}

// 5. Traverse and Display
void displayList(HeaderNode* h) {
    if (isEmpty(h)) {
        std::cout << "List is empty\n";
        return;
    }

    DataNode* current = h->next;

    std::cout << "List contents (" << h->size << " nodes): ";
    // Loop until we hit the header again
    while (current != (DataNode*)h) {
        std::cout << "[" << current->data << "] -> ";
        current = current->next;
    }
    std::cout << "(back to header)\n";
}

// 6. The Main Function to run it all
int main() {
    // Create our list at memory Address 100 (conceptually)
    HeaderNode* myList = createCircularHeaderList();

    std::cout << "Initial state:\n";
    displayList(myList);
    std::cout << "-----------------------\n";

    // Insert data
    insertAtBeginning(myList, 10);
    insertAtBeginning(myList, 20);
    insertAtBeginning(myList, 30);

    // Display after insertions
    // Note: Because we insert at the BEGINNING, 30 will be first!
    std::cout << "After inserting 10, 20, 30 at the beginning:\n";
    displayList(myList);

    return 0;
}
